```utils/pool.ts
import { consortiums, teams } from '../data/pool'
import type { ConsortiumStanding, Team } from '../types/pool'

export function formatCurrency(value: number, options?: { signed?: boolean; cents?: boolean }): string {
  const { signed = false, cents = false } = options ?? {}
  const abs = Math.abs(value)
  const body = abs.toLocaleString('en-US', {
    minimumFractionDigits: cents ? 2 : 0,
    maximumFractionDigits: cents ? 2 : 0,
  })
  const sign = value < 0 ? '−' : signed ? '+' : ''
  return `${sign}$${body}`
}

export function formatSigned(value: number): string {
  if (value === 0) return '—'
  return `${value > 0 ? '+' : '−'}${Math.abs(value)}`
}

export function ordinal(n: number): string {
  const s = ['th', 'st', 'nd', 'rd']
  const v = n % 100
  return `${n}${s[(v - 20) % 10] ?? s[v] ?? s[0]}`
}

export const pot = teams.reduce((sum, team) => sum + team.price, 0)
export const totalPoints = teams.reduce((sum, team) => sum + team.points, 0)
export const dollarsPerPoint = pot / totalPoints

export function teamById(id: string): Team {
  const found = teams.find((team) => team.id === id)
  if (!found) throw new Error(`Unknown team: ${id}`)
  return found
}

export function consortiumName(id: string): string {
  return consortiums.find((c) => c.id === id)?.name ?? id
}

export function buildStandings(): ConsortiumStanding[] {
  const rows = consortiums.map((consortium) => {
    const owned = teams
      .filter((team) => team.consortiumId === consortium.id)
      .sort((a, b) => b.points - a.points)
    const spend = owned.reduce((sum, team) => sum + team.price, 0)
    const points = owned.reduce((sum, team) => sum + team.points, 0)
    const weekDelta = owned.reduce((sum, team) => sum + team.weekDelta, 0)
    const value = points * dollarsPerPoint
    return {
      consortium,
      teams: owned,
      spend,
      points,
      value,
      net: value - spend,
      weekDelta,
      rank: 0,
    }
  })

  return rows
    .sort((a, b) => b.points - a.points || b.net - a.net)
    .map((row, index) => ({ ...row, rank: index + 1 }))
}

export function teamsByValue(): Team[] {
  return [...teams].sort((a, b) => b.points * dollarsPerPoint - b.price - (a.points * dollarsPerPoint - a.price))
}
```

```components/ViewBar.tsx
interface ViewBarProps<T extends string> {

  filterLabel?: string
  filterValue?: string
  filterOptions?: readonly string[]
  onFilterChange?: (value: string) => void

/**
 * Tabs and the period filter share one hairline row — the old layout gave the
 * filter a card of its own and pushed the data a full screen down.
 */
export function ViewBar<T extends string>({
  tabs,
  active,
  onChange,
  filterLabel,
  filterValue,
  filterOptions,
  onFilterChange,
}: ViewBarProps<T>) {
  return (
    <div className="flex flex-wrap items-end justify-between gap-x-8 gap-y-3 border-b border-rule">
      <div role="tablist" aria-label="View" className="flex gap-6">
        {tabs.map((tab) => {
          const isActive = tab.id === active
          return (
            <button
              key={tab.id}
              role="tab"
              type="button"
              aria-selected={isActive}
              onClick={() => onChange(tab.id)}
              className={[
                '-mb-px border-b-2 pb-3 text-[15px] transition-colors duration-150 ease-club',
                isActive
                  ? 'border-accent font-medium text-ink'

            >
              {tab.label}
            </button>
          )
        })}
      </div>

        <label className="flex items-center gap-2 pb-2.5">
          <span className="text-sm text-ink-faint">{filterLabel}</span>
          <span className="relative">
            <select
```

```components/CompareView.tsx
import React, { useState } from 'react'
import type { ConsortiumStanding } from '../types/pool'

interface CompareViewProps {

}

function Picker({
  label,
  value,
  rows,
  onChange,
}: {
  label: string
  value: string
  rows: ConsortiumStanding[]
  onChange: (id: string) => void
}) {
  return (
    <label className="block">
      <span className="eyebrow block text-ink-faint">{label}</span>
      <select
        value={value}
        onChange={(event) => onChange(event.target.value)}
        className="mt-2 w-full appearance-none border-b border-rule bg-transparent pb-2 font-display text-2xl font-semibold text-ink transition-colors duration-150 ease-club hover:border-rule-strong focus:border-accent focus:outline-none"
      >
        {rows.map((row) => (
          <option key={row.consortium.id} value={row.consortium.id}>
            {row.consortium.name}
          </option>
        ))}
      </select>
    </label>
  )
}

export function CompareView({ rows }: CompareViewProps) {
  const [leftId, setLeftId] = useState(rows[0]?.consortium.id ?? '')
  const [rightId, setRightId] = useState(rows[1]?.consortium.id ?? '')

  const left = rows.find((row) => row.consortium.id === leftId) ?? rows[0]
  const right = rows.find((row) => row.consortium.id === rightId) ?? rows[1]

  if (!left || !right) return null

  const lines: { label: string; left: string; right: string; leftWins: boolean }[] = [
    { label: 'Points', left: String(left.points), right: String(right.points), leftWins: left.points >= right.points },
    { label: 'This week', left: formatSigned(left.weekDelta), right: formatSigned(right.weekDelta), leftWins: left.weekDelta >= right.weekDelta },
    { label: 'Committed', left: formatCurrency(left.spend), right: formatCurrency(right.spend), leftWins: left.spend <= right.spend },
    { label: 'Holdings value', left: formatCurrency(Math.round(left.value)), right: formatCurrency(Math.round(right.value)), leftWins: left.value >= right.value },
    { label: 'Net position', left: formatCurrency(Math.round(left.net), { signed: true }), right: formatCurrency(Math.round(right.net), { signed: true }), leftWins: left.net >= right.net },
  ]

  return (
    <div>
      <div className="grid grid-cols-1 gap-8 sm:grid-cols-2">
        <Picker label="Challenger" value={left.consortium.id} rows={rows} onChange={setLeftId} />
        <Picker label="Opponent" value={right.consortium.id} rows={rows} onChange={setRightId} />
      </div>

      <dl className="mt-8">
        {lines.map((line) => (
          <div key={line.label} className="grid grid-cols-3 items-baseline border-b border-rule">
            <dd
              className={`nums row-y text-lg ${
                line.leftWins ? 'font-semibold text-ink' : 'text-ink-faint'
              }`}
            >
              {line.left}
            </dd>
            <dt className="row-y text-center text-[13px] text-ink-muted">{line.label}</dt>
            <dd
              className={`nums row-y text-right text-lg ${
                line.leftWins ? 'text-ink-faint' : 'font-semibold text-ink'
              }`}
            >
              {line.right}
            </dd>
          </div>
        ))}
      </dl>

      <p className="mt-5 font-display text-[15px] italic text-ink-faint">
        {left.points === right.points
          ? 'Dead level. Settle it at the bar.'
          : `${left.points > right.points ? left.consortium.name : right.consortium.name} by ${Math.abs(
              left.points - right.points,
            )} points — a lead, not a legacy.`}
      </p>
    </div>
  )
}
```

```data/styleGuide.ts
## 9. Motion

- Library: framer-motion. Always honor \`prefers-reduced-motion\`.
```

```pages/Handoff.tsx
import React, { useState } from 'react'
import { CheckIcon, CopyIcon, DownloadIcon } from 'lucide-react'
import { PageHeader } from '../components/PageHeader'
import { styleGuide } from '../data/styleGuide'

const FILE_NAME = 'THE-CALCUTTA-STYLE-GUIDE.md'

export function Handoff() {
  const [copied, setCopied] = useState(false)

  const handleCopy = async () => {
    try {
      await navigator.clipboard.writeText(styleGuide)
      setCopied(true)
      window.setTimeout(() => setCopied(false), 2000)
    } catch {
      setCopied(false)
    }
  }

  const handleDownload = () => {
    const blob = new Blob([styleGuide], { type: 'text/markdown;charset=utf-8' })
    const url = URL.createObjectURL(blob)
    const link = document.createElement('a')
    link.href = url
    link.download = FILE_NAME
    document.body.appendChild(link)
    link.click()
    document.body.removeChild(link)
    URL.revokeObjectURL(url)
  }

  const lineCount = styleGuide.split('\n').length

  return (
    <div className="space-y-10">
      <PageHeader
        eyebrow="Handoff"
        title="Style guide"
        lede="Everything an implementing agent needs, and nothing it can misread as a suggestion."
        aside={
          <div className="flex gap-2">
            <button
              type="button"
              onClick={handleCopy}
              className="flex items-center gap-2 rounded-sm border border-ink bg-ink px-3.5 py-2 text-sm font-medium text-paper transition-colors duration-150 ease-club hover:bg-ink/90"
            >
              {copied ? <CheckIcon className="h-4 w-4" /> : <CopyIcon className="h-4 w-4" />}
              {copied ? 'Copied' : 'Copy markdown'}
            </button>
            <button
              type="button"
              onClick={handleDownload}
              className="flex items-center gap-2 rounded-sm border border-rule bg-surface px-3.5 py-2 text-sm text-ink transition-colors duration-150 ease-club hover:border-rule-strong"
            >
              <DownloadIcon className="h-4 w-4" />
              Download .md
            </button>
          </div>
        }
      />

      <section aria-label="How to use this">
        <h2 className="eyebrow border-b border-ink pb-2.5 text-ink">How to hand it over</h2>
        <ol className="mt-4 space-y-2.5 text-[15px] text-ink-muted">
          <li>
            1. Copy the markdown, or download it as <span className="text-ink">{FILE_NAME}</span>.
          </li>
          <li>
            2. In Replit, save it to the repo root and tell the agent: “Read
            {' '}{FILE_NAME} and follow it exactly for all UI work. Where it conflicts with your
            defaults, it wins.”
          </li>
          <li>
            3. Sections 5 and 8 are drop-in code — the agent should paste the tokens and the logo
            verbatim rather than reinterpreting them.
          </li>
          <li>
            4. Section 12 is the review checklist. Ask the agent to run it before every handback.
          </li>
        </ol>
      </section>

      <section aria-label="Style guide source">
        <div className="flex items-baseline justify-between border-b border-ink pb-2.5">
          <h2 className="eyebrow text-ink">The document</h2>
          <span className="nums text-[13px] text-ink-faint">{lineCount} lines · markdown</span>
        </div>
        <pre className="mt-5 max-h-[620px] overflow-auto whitespace-pre-wrap break-words border border-rule bg-surface p-6 text-[13px] leading-relaxed text-ink-muted">
          {styleGuide}
        </pre>
      </section>
    </div>
  )
}
```
