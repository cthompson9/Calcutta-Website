```App.tsx
import React, { useState } from 'react'
import { AnimatePresence, motion } from 'framer-motion'
import { PhoneShell } from './components/PhoneShell'
import { TopBar } from './components/TopBar'
import { BottomTabs, type TabKey } from './components/BottomTabs'
import { Results } from './pages/Results'
import { M2MTracker } from './pages/M2MTracker'
import { Trades } from './pages/Trades'
import { AuctionResults } from './pages/AuctionResults'
import type { Season } from './types/calcutta'

export function App() {
  const [season, setSeason] = useState<Season>('2026')
  const [tab, setTab] = useState<TabKey>('results')

  return (
    <PhoneShell
      header={<TopBar season={season} onSeasonChange={setSeason} />}
      tabBar={<BottomTabs active={tab} onChange={setTab} />}
    >
      <AnimatePresence mode="wait" initial={false}>
        <motion.div
          key={`${tab}-${season}`}
          initial={{ opacity: 0, y: 8 }}
          animate={{ opacity: 1, y: 0 }}
          exit={{ opacity: 0 }}
          transition={{ duration: 0.2, ease: [0.23, 1, 0.32, 1] }}
        >
          {tab === 'results' && <Results season={season} />}
          {tab === 'tracker' && <M2MTracker season={season} />}
          {tab === 'trades' && <Trades season={season} />}
          {tab === 'auction' && <AuctionResults season={season} />}
        </motion.div>
      </AnimatePresence>
    </PhoneShell>
  )
}
```

```utils/format.ts
export function currency(value: number, compact = false): string {
  if (compact && Math.abs(value) >= 1000) {
    return `$${(value / 1000).toFixed(value % 1000 === 0 ? 0 : 1)}k`
  }
  return `$${Math.round(value).toLocaleString('en-US')}`
}

export function signedCurrency(value: number, compact = false): string {
  const sign = value > 0 ? '+' : value < 0 ? '-' : ''
  return `${sign}${currency(Math.abs(value), compact)}`
}

export function percent(value: number): string {
  const sign = value > 0 ? '+' : value < 0 ? '−' : ''
  return `${sign}${Math.abs(value).toFixed(1)}%`
}

export function decimal(value: number, places = 2): string {
  return value.toFixed(places)
}

export function toneClass(value: number): string {
  if (value > 0) return 'text-up'
  if (value < 0) return 'text-down'
  return 'text-muted'
}
```

```components/CompareView.tsx
import React, { useState } from 'react'
import type { Consortium } from '../types/calcutta'
import { SheetSelect } from './SheetSelect'
import { currency, percent, decimal, toneClass } from '../utils/format'

interface CompareViewProps {
  consortiums: Consortium[]
}

interface Metric {
  label: string
  render: (c: Consortium) => string
  tone?: (c: Consortium) => string
  weight: (c: Consortium) => number
}

const metrics: Metric[] = [
  {
    label: 'MTM return',
    render: (c) => percent(c.mtmReturn),
    tone: (c) => toneClass(c.mtmReturn),
    weight: (c) => Math.abs(c.mtmReturn),
  },
  {
    label: 'Realized',
    render: (c) => percent(c.realizedReturn),
    tone: (c) => toneClass(c.realizedReturn),
    weight: (c) => Math.abs(c.realizedReturn),
  },
  {
    label: 'Exposure',
    render: (c) => currency(c.exposure),
    weight: (c) => c.exposure,
  },
  {
    label: 'Teams',
    render: (c) => decimal(c.teams),
    weight: (c) => c.teams,
  },
]

export function CompareView({ consortiums }: CompareViewProps) {
  const [leftName, setLeftName] = useState(consortiums[0]?.name ?? '')
  const [rightName, setRightName] = useState(consortiums[1]?.name ?? '')

  const names = consortiums.map((c) => c.name)
  const left = consortiums.find((c) => c.name === leftName) ?? consortiums[0]
  const right = consortiums.find((c) => c.name === rightName) ?? consortiums[1]

  if (!left || !right) {
    return (
      <p className="px-4 py-10 text-center text-sm text-muted">
        Not enough consortiums in this season to compare.
      </p>
    )
  }

  return (
    <div className="px-4 pb-6 pt-4">
      <div className="grid grid-cols-2 gap-2">
        <SheetSelect
          label="A"
          value={left.name}
          options={names}
          onChange={setLeftName}
        />
        <SheetSelect
          label="B"
          value={right.name}
          options={names}
          onChange={setRightName}
        />
      </div>

      <dl className="mt-5">
        {metrics.map((metric) => {
          const l = metric.weight(left)
          const r = metric.weight(right)
          const total = l + r || 1
          const leftPct = Math.max(6, Math.min(94, (l / total) * 100))
          return (
            <div key={metric.label} className="border-t border-line py-3.5">
              <dt className="label-mono text-2xs text-muted">{metric.label}</dt>
              <dd className="mt-2">
                <div className="flex items-baseline justify-between gap-3">
                  <span
                    className={`font-mono text-[15px] tabular-nums ${metric.tone ? metric.tone(left) : 'text-ink'}`}
                  >
                    {metric.render(left)}
                  </span>
                  <span
                    className={`font-mono text-[15px] tabular-nums ${metric.tone ? metric.tone(right) : 'text-ink'}`}
                  >
                    {metric.render(right)}
                  </span>
                </div>
                <div className="mt-2 flex h-1.5 gap-1 overflow-hidden rounded-full">
                  <span
                    className="h-full rounded-full bg-ink transition-[width] duration-300 ease-smooth"
                    style={{ width: `${leftPct}%` }}
                  />
                  <span
                    className="h-full flex-1 rounded-full bg-line"
                  />
                </div>
              </dd>
            </div>
          )
        })}
      </dl>
    </div>
  )
}
```

```pages/Results.tsx
              key={consortium.id}

        </ul>
      )}

  )
```

```components/TrackerChart.tsx
import React from 'react'
import type { TrackerSeries } from '../types/calcutta'
import { trackerPeriods } from '../data/tracker'

interface TrackerChartProps {
  series: TrackerSeries[]
  activeId: string
}

const WIDTH = 320
const HEIGHT = 150
const PAD_X = 4
const PAD_Y = 10

export function TrackerChart({ series, activeId }: TrackerChartProps) {
  const all = series.flatMap((s) => s.points.map((p) => p.value))
  const max = Math.max(...all, 5)
  const min = Math.min(...all, -5)

  const x = (index: number) =>
    PAD_X + (index / (trackerPeriods.length - 1)) * (WIDTH - PAD_X * 2)
  const y = (value: number) =>
    PAD_Y + ((max - value) / (max - min)) * (HEIGHT - PAD_Y * 2)

  const path = (s: TrackerSeries) =>
    s.points.map((p, i) => `${i === 0 ? 'M' : 'L'}${x(i)},${y(p.value)}`).join(' ')

  const active = series.find((s) => s.id === activeId)

  return (
    <figure className="px-4">
      <svg
        viewBox={`0 0 ${WIDTH} ${HEIGHT}`}
        className="w-full"
        role="img"
        aria-label={`Cumulative mark-to-market return by period. ${active?.name ?? ''} highlighted.`}
      >
        <line
          x1={PAD_X}
          x2={WIDTH - PAD_X}
          y1={y(0)}
          y2={y(0)}
          stroke="#E6E3DD"
          strokeWidth="1"
        />
        {series
          .filter((s) => s.id !== activeId)
          .map((s) => (
            <path
              key={s.id}
              d={path(s)}
              fill="none"
              stroke="#DDD9D2"
              strokeWidth="1.25"
              strokeLinecap="round"
            />
          ))}
        {active && (
          <>
            <path
              d={path(active)}
              fill="none"
              stroke="#0B0B0B"
              strokeWidth="2"
              strokeLinecap="round"
              strokeLinejoin="round"
            />
            {active.points.map((p, i) => (
              <circle
                key={p.period}
                cx={x(i)}
                cy={y(p.value)}
                r={i === active.points.length - 1 ? 3.5 : 2}
                fill="#0B0B0B"
              />
            ))}
          </>
        )}
      </svg>
      <figcaption className="mt-1 flex justify-between">
        {trackerPeriods.map((p) => (
          <span key={p} className="label-mono text-2xs text-muted">
            {p}
          </span>
        ))}
      </figcaption>
    </figure>
  )
}
```

```pages/AuctionResults.tsx
import React, { useState } from 'react'
import { SearchIcon } from 'lucide-react'
import { PageHeading } from '../components/PageHeading'
import { Segmented } from '../components/Segmented'
import { auctionLots, auctionSummary } from '../data/auction'
import { currency } from '../utils/format'
import type { Season } from '../types/calcutta'

type AuctionSort = 'price' | 'order' | 'buyer'

interface AuctionResultsProps {
  season: Season
}

export function AuctionResults({ season }: AuctionResultsProps) {
  const [sort, setSort] = useState<AuctionSort>('price')
  const [query, setQuery] = useState('')

  const filtered = auctionLots.filter((lot) => {
    const q = query.trim().toLowerCase()
    if (!q) return true
    return (
      lot.name.toLowerCase().includes(q) ||
      lot.abbr.toLowerCase().includes(q) ||
      lot.buyer.toLowerCase().includes(q)
    )
  })

  const sorted = [...filtered].sort((a, b) => {
    if (sort === 'price') return b.price - a.price
    if (sort === 'order') return a.order - b.order
    return a.buyer.localeCompare(b.buyer)
  })

  const maxPrice = Math.max(...auctionLots.map((l) => l.price))

  return (
    <div className="pb-8">
      <PageHeading
        title="Auction Results"
        meta={`${auctionSummary.date} · ${auctionSummary.lots} lots · ${season}`}
      >
        <div className="mt-4 flex items-baseline gap-2">

            {currency(auctionSummary.totalPot)}
          </span>
          <span className="label-mono text-2xs text-muted">
```
