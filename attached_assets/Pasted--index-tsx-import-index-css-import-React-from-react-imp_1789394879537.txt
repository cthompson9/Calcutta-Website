```utils/format.ts
export function money(value: number): string {
  const abs = Math.abs(Math.round(value))
  return `${value < 0 ? '−' : ''}$${abs.toLocaleString('en-US')}`
}

export function moneySigned(value: number): string {
  const abs = Math.abs(Math.round(value))
  const sign = value > 0 ? '+' : value < 0 ? '−' : ''
  return `${sign}$${abs.toLocaleString('en-US')}`
}

export function moneyCompact(value: number): string {
  const abs = Math.abs(value)
  const sign = value > 0 ? '+' : value < 0 ? '−' : ''
  if (abs >= 1000) {
    const k = abs / 1000
    return `${sign}$${k % 1 === 0 ? k.toFixed(0) : k.toFixed(1)}k`
  }
  return abs === 0 ? '0' : `${sign}$${Math.round(abs)}`
}

export function percentSigned(value: number): string {
  const sign = value > 0 ? '+' : value < 0 ? '−' : ''
  return `${sign}${Math.abs(value * 100).toFixed(0)}%`
}
```

```utils/scale.ts
import type { Lot, ScaleMode } from '../types/lot'

const THRESHOLD = 300

/** Symlog-style transform: linear near zero, logarithmic in the tails. */
export function transform(value: number, mode: ScaleMode): number {
  if (mode === 'even') return value
  return Math.sign(value) * Math.log1p(Math.abs(value) / THRESHOLD)
}

const TICK_CANDIDATES = [
  0, 250, 500, 1000, 1500, 2000, 3000, 4000, 5000, 6000, 8000, 10000, 15000, 20000,
]

export interface YScale {
  min: number
  max: number
  toY: (value: number) => number
  ticks: number[]
}

export function buildYScale(
  values: number[],
  mode: ScaleMode,
  top: number,
  bottom: number,
): YScale {
  const rawMin = Math.min(0, ...values)
  const rawMax = Math.max(0, ...values)
  const span = rawMax - rawMin || 1
  const min = rawMin - span * 0.06
  const max = rawMax + span * 0.06

  const tMin = transform(min, mode)
  const tMax = transform(max, mode)
  const tSpan = tMax - tMin || 1

  const toY = (value: number) => bottom - ((transform(value, mode) - tMin) / tSpan) * (bottom - top)

  const ticks: number[] = []
  for (const candidate of TICK_CANDIDATES) {
    if (candidate === 0) {
      ticks.push(0)
      continue
    }
    if (candidate <= max) ticks.push(candidate)
    if (-candidate >= min) ticks.push(-candidate)
  }

  // Thin out ticks that would collide vertically.
  const sorted = ticks.sort((a, b) => b - a)
  const kept: number[] = []
  for (const tick of sorted) {
    const y = toY(tick)
    if (tick !== 0 && kept.some((k) => Math.abs(toY(k) - y) < 26)) continue
    kept.push(tick)
  }

  return { min, max, toY, ticks: kept }
}

export function sliceValues(lots: Lot[], start: number): number[] {
  const out: number[] = []
  for (const lot of lots) {
    for (let i = start; i < lot.series.length; i++) out.push(lot.series[i])
  }
  return out
}

export function currentValue(lot: Lot): number {
  return lot.series[lot.series.length - 1]
}

export function weekMove(lot: Lot): number {
  const n = lot.series.length
  return n < 2 ? 0 : lot.series[n - 1] - lot.series[n - 2]
}

export function returnRate(lot: Lot): number {
  return currentValue(lot) / lot.price
}

/** Dot radius encoding auction price, kept in a legible range. */
export function priceRadius(price: number, base = 3): number {
  return base + Math.sqrt(price) / 22
}
```

```hooks/useMeasure.ts
import { useEffect, useRef, useState } from 'react'

export function useMeasure<T extends HTMLElement>() {
  const ref = useRef<T | null>(null)
  const [size, setSize] = useState({ width: 0, height: 0 })

  useEffect(() => {
    const node = ref.current
    if (!node) return
    const observer = new ResizeObserver((entries) => {
      const rect = entries[0].contentRect
      setSize({ width: rect.width, height: rect.height })
    })
    observer.observe(node)
    setSize({ width: node.clientWidth, height: node.clientHeight })
    return () => observer.disconnect()
  }, [])

  return { ref, ...size }
}
```

```components/FocusChart.tsx
import React, { useMemo, useState } from 'react'
import type { Lot, ScaleMode } from '../types/lot'
import { useMeasure } from '../hooks/useMeasure'
import { buildYScale, priceRadius, sliceValues } from '../utils/scale'
import { moneyCompact, moneySigned } from '../utils/format'

interface FocusChartProps {
  lots: Lot[]
  labels: string[]
  start: number
  scaleMode: ScaleMode
  pinned: string[]
  hovered: string | null
  onHover: (id: string | null) => void
  onTogglePin: (id: string) => void
}

const MARGIN = { top: 18, right: 64, bottom: 30, left: 62 }

export function FocusChart({
  lots,
  labels,
  start,
  scaleMode,
  pinned,
  hovered,
  onHover,
  onTogglePin,
}: FocusChartProps) {
  const { ref, width, height } = useMeasure<HTMLDivElement>()
  const [crosshair, setCrosshair] = useState<number | null>(null)

  const weeks = labels.length - start
  const innerW = Math.max(width - MARGIN.left - MARGIN.right, 10)
  const bottom = height - MARGIN.bottom

  const scale = useMemo(
    () => buildYScale(sliceValues(lots, start), scaleMode, MARGIN.top, Math.max(bottom, 40)),
    [lots, start, scaleMode, bottom],
  )

  const x = (index: number) => MARGIN.left + ((index - start) / Math.max(weeks - 1, 1)) * innerW

  const pathFor = (lot: Lot) =>
    lot.series
      .slice(start)
      .map((v, i) => `${i === 0 ? 'M' : 'L'}${x(start + i).toFixed(1)},${scale.toY(v).toFixed(1)}`)
      .join(' ')

  const pinnedSet = new Set(pinned)
  const emphasized = lots.filter((l) => pinnedSet.has(l.id) || l.id === hovered)
  const background = lots.filter((l) => !pinnedSet.has(l.id) && l.id !== hovered)

  const colorFor = (lot: Lot) => {
    if (lot.id === hovered) return '#1B1A16'
    return lot.series[lot.series.length - 1] >= 0 ? '#2E6B4F' : '#B0472C'
  }

  const readout = crosshair === null ? null : crosshair

  const handleMove = (event: React.MouseEvent<SVGSVGElement>) => {
    const rect = event.currentTarget.getBoundingClientRect()
    const px = event.clientX - rect.left
    const ratio = (px - MARGIN.left) / Math.max(innerW, 1)
    const index = Math.round(ratio * (weeks - 1)) + start
    setCrosshair(Math.min(labels.length - 1, Math.max(start, index)))
  }

  return (
    <div ref={ref} className="relative h-[300px] w-full sm:h-[380px] lg:h-[460px]">
      {width > 0 && height > 0 && (
        <svg
          width={width}
          height={height}
          role="img"
          aria-label="Net lot value by week for all 32 lots"
          onMouseMove={handleMove}
          onMouseLeave={() => {
            setCrosshair(null)
            onHover(null)
          }}
        >
          {scale.ticks.map((tick) => {
            const y = scale.toY(tick)
            const isZero = tick === 0
            return (
              <g key={tick}>
                <line
                  x1={MARGIN.left}
                  x2={width - MARGIN.right}
                  y1={y}
                  y2={y}
                  stroke={isZero ? '#1B1A16' : '#E3DDCD'}
                  strokeWidth={isZero ? 1 : 1}
                  strokeDasharray={isZero ? '0' : '2 4'}
                  opacity={isZero ? 0.45 : 1}
                />
                <text
                  x={MARGIN.left - 10}
                  y={y + 3.5}
                  textAnchor="end"
                  className="font-mono"
                  fontSize={10}
                  fill={isZero ? '#1B1A16' : '#6F6A5C'}
                >
                  {isZero ? 'break even' : moneyCompact(tick)}
                </text>
              </g>
            )
          })}

          {labels.slice(start).map((label, i) => {
            const index = start + i
            const show = weeks <= 14 || i % 2 === 0 || index === labels.length - 1
            if (!show) return null
            return (
              <text
                key={label}
                x={x(index)}
                y={height - 10}
                textAnchor="middle"
                className="font-mono"
                fontSize={10}
                fill={readout === index ? '#1B1A16' : '#6F6A5C'}
              >
                {label}
              </text>
            )
          })}

          {readout !== null && (
            <line
              x1={x(readout)}
              x2={x(readout)}
              y1={MARGIN.top - 6}
              y2={bottom}
              stroke="#1B1A16"
              strokeWidth={1}
              opacity={0.2}
            />
          )}

          {background.map((lot) => (
            <path
              key={lot.id}
              d={pathFor(lot)}
              fill="none"
              stroke="#CFC8B4"
              strokeWidth={1}
              strokeLinecap="round"
            />
          ))}

          {emphasized.map((lot) => {
            const color = colorFor(lot)
            const last = lot.series[lot.series.length - 1]
            return (
              <g key={lot.id}>
                <path
                  d={pathFor(lot)}
                  fill="none"
                  stroke={color}
                  strokeWidth={2}
                  strokeLinecap="round"

                  strokeWidth={1.5}

                <text
                  x={width - MARGIN.right + 10}

                  className="font-mono"
                  fontSize={10}
                  fill={color}
                >
                  {lot.abbr}
                </text>
              </g>
            )
          })}

          {lots.map((lot) => (
            <path
              key={`hit-${lot.id}`}
              d={pathFor(lot)}
              fill="none"
              stroke="transparent"
              strokeWidth={12}
              style={{ cursor: 'pointer' }}
              onMouseEnter={() => onHover(lot.id)}
              onClick={() => onTogglePin(lot.id)}

          ))}
        </svg>

      {readout !== null && emphasized.length > 0 && (
        <div
          className="pointer-events-none absolute top-4 rounded-sm border border-rule bg-paper/95 px-3 py-2 shadow-sm"

            left: Math.min(Math.max(x(readout) + 12, 8), Math.max(width - 190, 8)),

                    {moneySigned(lot.series[readout])}

                </li>
              ))}
          </ul>
        </div>
      )}
    </div>
  )
}
```

```components/LotGrid.tsx
interface LotGridProps {

  scaleMode: ScaleMode
  pinned: string[]
  hovered: string | null
  onHover: (id: string | null) => void
  onSelect: (id: string) => void
}

const W = 132
const H = 46

export function LotGrid({ lots, start, scaleMode, pinned, hovered, onHover, onSelect }: LotGridProps) {
  const scale = useMemo(
    () => buildYScale(sliceValues(lots, start), scaleMode, 4, H - 4),
    [lots, start, scaleMode],
  )

  const ordered = useMemo(
    () => [...lots].sort((a, b) => currentValue(b) - currentValue(a)),
    [lots],
  )

  const pinnedSet = new Set(pinned)
  const weeks = lots[0].series.length - start
  const x = (i: number) => (i / Math.max(weeks - 1, 1)) * W

  return (
    <div className="grid grid-cols-2 gap-px bg-rule sm:grid-cols-3 lg:grid-cols-4 xl:grid-cols-6">
      {ordered.map((lot) => {
        const value = currentValue(lot)
        const color = value >= 0 ? '#2E6B4F' : '#B0472C'

        const path = points
          .map((v, i) => `${i === 0 ? 'M' : 'L'}${x(i).toFixed(1)},${scale.toY(v).toFixed(1)}`)
          .join(' ')
        const area = `${path} L${x(points.length - 1).toFixed(1)},${scale.toY(0).toFixed(1)} L0,${scale
          .toY(0)
          .toFixed(1)} Z`
        const active = pinnedSet.has(lot.id) || lot.id === hovered

        return (
          <button
            key={lot.id}
            type="button"
            onMouseEnter={() => onHover(lot.id)}
            onMouseLeave={() => onHover(null)}
            onClick={() => onSelect(lot.id)}
            className={`group flex flex-col gap-1 px-3 py-3 text-left transition-colors duration-150 ease-out ${
              active ? 'bg-[#F1ECDF]' : 'bg-paper hover:bg-[#F5F1E6]'
            }`}
          >
            <div className="flex items-baseline justify-between gap-2">
              <span className="font-mono text-[11px] tracking-wider text-ink">{lot.abbr}</span>
              <span className="font-mono text-[11px] tabular-nums" style={{ color }}>
                {moneySigned(value)}
              </span>
            </div>
            <svg viewBox={`0 0 ${W} ${H}`} width="100%" height={H} preserveAspectRatio="none" aria-hidden>
              <line x1={0} x2={W} y1={scale.toY(0)} y2={scale.toY(0)} stroke="#E3DDCD" strokeWidth={1} />
              <path d={area} fill={color} fillOpacity={0.08} />
              <path d={path} fill="none" stroke={color} strokeWidth={1.5} strokeLinejoin="round" />
              <circle cx={x(points.length - 1)} cy={scale.toY(points[points.length - 1])} r={2.5} fill={color} />
            </svg>
            <span className="truncate text-[11px] text-muted">
              {lot.name} · ${lot.price.toLocaleString('en-US')}
            </span>
          </button>
        )
      })}
    </div>
  )
}
```

```components/SummaryBar.tsx
interface SummaryBarProps {
```

```pages/LotValues.tsx
import React, { useEffect, useMemo, useState } from 'react'
import { XIcon } from 'lucide-react'
import type { ScaleMode } from '../types/lot'
import { lots, POOL_SIZE, WEEK_LABELS } from '../data/lots'
import { currentValue } from '../utils/scale'
import { FocusChart } from '../components/FocusChart'
import { LotGrid } from '../components/LotGrid'
import { SummaryBar } from '../components/SummaryBar'
import { Segmented } from '../components/Segmented'

export type ViewMode = 'focus' | 'grid'
export type RangeMode = 'season' | 'recent'

interface LotValuesProps {
  view: ViewMode
  scale: ScaleMode
}

const RECENT_WEEKS = 6

function defaultPinned(): string[] {
  const ordered = [...lots].sort((a, b) => currentValue(b) - currentValue(a))
  return [ordered[0].id, ordered[1].id, ordered[ordered.length - 2].id, ordered[ordered.length - 1].id]
}

export function LotValues({ view, scale }: LotValuesProps) {
  const [viewMode, setViewMode] = useState<ViewMode>(view)
  const [scaleMode, setScaleMode] = useState<ScaleMode>(scale)
  const [range, setRange] = useState<RangeMode>('season')
  const [pinned, setPinned] = useState<string[]>(defaultPinned)
  const [hovered, setHovered] = useState<string | null>(null)

  useEffect(() => setViewMode(view), [view])
  useEffect(() => setScaleMode(scale), [scale])

  const start = range === 'recent' ? Math.max(WEEK_LABELS.length - RECENT_WEEKS - 1, 0) : 0
  const pinnedLots = useMemo(() => lots.filter((l) => pinned.includes(l.id)), [pinned])

  const togglePin = (id: string) =>
    setPinned((current) =>
      current.includes(id) ? current.filter((p) => p !== id) : [...current, id],
    )

  return (
    <main className="min-h-full w-full bg-cream">
      <div className="mx-auto flex max-w-[1440px] flex-col gap-6 px-5 py-10 sm:px-8 lg:py-14">
        <header className="max-w-3xl">
          <p className="label-mono text-muted">Calcutta XII · NFL 2026</p>
          <h1 className="mt-2 font-display text-4xl leading-none text-ink sm:text-5xl lg:text-6xl">
            Lot values over time
          </h1>
          <p className="mt-3 text-sm leading-relaxed text-muted">
            Each line is one of the 32 lots: expected payout normalized to the {' '}
            <span className="text-ink">${POOL_SIZE.toLocaleString('en-US')}</span> pool, minus what it
            sold for at auction. Above the line, the lot is worth more than it cost. Dot size at the
            end of a highlighted line reflects auction price.
          </p>
        </header>

        <SummaryBar lots={lots} poolSize={POOL_SIZE} weekLabel={WEEK_LABELS[WEEK_LABELS.length - 1]} />

        <div className="flex flex-wrap items-center gap-x-6 gap-y-3">
          <Segmented
            label="View"
            value={viewMode}
            onChange={setViewMode}
            options={[
              { value: 'focus', label: 'Compare' },
              { value: 'grid', label: 'All 32' },
            ]}
          />
          <Segmented
            label="Axis"
            value={scaleMode}
            onChange={setScaleMode}
            options={[
              { value: 'compressed', label: 'Compressed' },
              { value: 'even', label: 'Even' },
            ]}
          />
          <Segmented
            label="Range"
            value={range}
            onChange={setRange}
            options={[
              { value: 'season', label: 'Season' },
              { value: 'recent', label: 'Last 6' },
            ]}
          />
          <p className="text-xs text-muted">
            {scaleMode === 'compressed'
              ? 'Tails compressed so the pack near break-even stays readable.'
              : 'True dollar spacing — outliers dominate the axis.'}
          </p>
        </div>

        {viewMode === 'focus' && (
          <div className="flex flex-wrap items-center gap-2">
            <span className="label-mono text-muted">Highlighted</span>
            {pinnedLots.length === 0 && (
              <span className="text-xs text-muted">
                Click any line or any row to bring a lot forward.
              </span>
            )}
            {pinnedLots.map((lot) => (
              <button
                key={lot.id}
                type="button"
                onClick={() => togglePin(lot.id)}
                className="group flex items-center gap-1.5 border border-rule bg-paper py-1 pl-2 pr-1.5 transition-colors duration-150 ease-out hover:border-ink"
              >
                <span
                  className="h-1.5 w-1.5 rounded-full"
                  style={{ backgroundColor: currentValue(lot) >= 0 ? '#2E6B4F' : '#B0472C' }}
                />
                <span className="font-mono text-[11px] text-ink">{lot.abbr}</span>
                <XIcon className="h-3 w-3 text-muted transition-colors duration-150 ease-out group-hover:text-ink" />
              </button>
            ))}
            {pinnedLots.length > 0 && (
              <button
                type="button"
                onClick={() => setPinned([])}
                className="font-mono text-[10px] uppercase tracking-wider text-muted underline-offset-4 hover:text-ink hover:underline"
              >
                Clear
              </button>
            )}
          </div>
        )}

        {viewMode === 'focus' ? (
          <div className="border border-rule bg-paper p-3 sm:p-5">
            <FocusChart
              lots={lots}
              labels={WEEK_LABELS}
              start={start}
              scaleMode={scaleMode}
              pinned={pinned}
              hovered={hovered}
              onHover={setHovered}
              onTogglePin={togglePin}
            />
          </div>
        ) : (
          <div className="border border-rule bg-rule">
            <LotGrid
              lots={lots}
              start={start}
              scaleMode={scaleMode}
              pinned={pinned}
              hovered={hovered}
              onHover={setHovered}
              onSelect={(id) => {
                setPinned([id])
                setViewMode('focus')
              }}
            />
          </div>
        )}

        <p className="text-xs text-muted">
          {viewMode === 'focus'
            ? 'Hover a line for its weekly value; click to keep it highlighted. Everything else stays as light context.'
            : 'Every lot on a shared axis, ranked by current net value. Click one to open it in the comparison view.'}
        </p>
      </div>
    </main>
  )
}
```
